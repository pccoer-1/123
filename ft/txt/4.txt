# ✅ 7. CLEAN PYTHON IMPLEMENTATION

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn import linear_model
from sklearn.model_selection import train_test_split

data_url = "http://lib.stat.cmu.edu/datasets/boston"
raw_df = pd.read_csv(data_url, sep=r"\s+", skiprows=22, header=None)

data = np.hstack([raw_df.values[::2, :], raw_df.values[1::2, :2]])
target = raw_df.values[1::2, 2]

feature_names = np.array(['CRIM','ZN','INDUS','CHAS','NOX','RM','AGE','DIS','RAD','TAX','PTRATIO','B','LSTAT'])

boston = {'data': data, 'target': target, 'feature_names': feature_names}

x = pd.DataFrame(boston['data'], columns=boston['feature_names'])
y = pd.DataFrame(boston['target'])

plt.xlabel('RM')
plt.ylabel('target')
plt.scatter(boston['data'][:,5], boston['target'], color='red', marker='+')
plt.show()

reg = linear_model.LinearRegression()

x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.20, random_state=42)

reg.fit(x_train, y_train)

print(reg.coef_)

print(reg.score(x_test, y_test))

y_pred = reg.predict(x_test)

print(y_pred)
```
